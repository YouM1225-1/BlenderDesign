    if not temp.is_absolute() or temp != Path(os.path.realpath(temp)) or Path(os.path.realpath(raw)) != temp or not stat.S_ISDIR(before.st_mode) or stat.S_ISLNK(before.st_mode) or before.st_uid != os.getuid() or stat.S_IMODE(before.st_mode) != 0o700:
        raise RuntimeError("unsafe canonical Blender tempdir")
    if not scratch.is_absolute() or scratch != Path(os.path.realpath(scratch)) or scratch.parent != temp or scratch.name != "blender_mcp" or os.path.lexists(scratch):
        raise RuntimeError("unsafe Blender MCP scratch child")
    if identity(os.lstat(temp)) != expected:
        raise RuntimeError("Blender tempdir changed")
    print(f"R10_SCRATCH_GREEN t_dev={before.st_dev} t_ino={before.st_ino}")
    return 0
